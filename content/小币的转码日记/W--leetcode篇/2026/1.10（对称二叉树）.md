```python
class Solution:

    def isSymmetric(self, root: Optional[TreeNode]) -> bool:

        is_symmetric = True

        left = []

        right = []

        def inorder(root):

            if not root:

                left.append(None)

                return

            left.append(root.val)

            inorder(root.left)

            inorder(root.right)

        def rearorder(root):

            if not root:

                right.append(None)

                return

            right.append(root.val)

            rearorder(root.right)

            rearorder(root.left)

        inorder(root.left)

        rearorder(root.right)

        if left != right:

            is_symmetric = False

        return is_symmetric
```
我的思路是对根节点的左子节点和右子节点分别进行前序遍历，只不过在添加的时候左右顺序变化了一下，注意这里要把None节点也加进去，通过None节点能够排除很多特殊的位置
```python
class Solution:

    def isSymmetric(self, root: Optional[TreeNode]) -> bool:

        # 辅助函数：同步对比一对对称位置的节点

        def check(left: Optional[TreeNode], right: Optional[TreeNode]) -> bool:

            # 条件1：都为空 → 对称

            if not left and not right:

                return True

            # 条件2：一个空一个非空 → 不对称

            if not left or not right:

                return False

            # 条件3：值不等 → 不对称；条件4：值相等则递归对比子节点

            return left.val == right.val and check(left.left, right.right) and check(left.right, right.left)

        # 空树直接对称；否则对比根的左右子树

        return check(root.left, root.right) if root else True
```
