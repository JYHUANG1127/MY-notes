```python
class Solution:

    def copyRandomList(self, head: 'Optional[Node]') -> 'Optional[Node]':

        if not head:

            return None  

        old_to_new = {}

        cur = head

        while cur:

            old_to_new[cur] = Node(cur.val) 

            cur = cur.next
        #这里关键的是这里，这个时候dic中的key是cur，也就是原节点，而val是新节点，这个key方便后面从dic中找到它对应的新节点
        

        # 2. 给新节点的next和random赋值

        cur = head

        while cur:

            new_node = old_to_new[cur]

            # next指针：原节点next对应的新节点（没有则为None）

            new_node.next = old_to_new.get(cur.next, None)

            # random指针：原节点random对应的新节点（没有则为None）

            new_node.random = old_to_new.get(cur.random, None)

            cur = cur.next

        # 3. 返回新链表的头节点

        return old_to_new[head]
```
豆包的方法是穿插法，在原链表中的每一个节点与它下一个节点之间在新建立一个节点，要求这个节点与上一个节点的值相同，然后random指针也与它相同，最后拆成原链表和新链表即可
```python
class Solution:

    def copyRandomList(self, head: 'Node') -> 'Node':

        if not head:

            return None

        # 1. 穿插新节点：原节点cur后插入新节点cur.next

        cur = head

        while cur:

            new_node = Node(cur.val)

            new_node.next = cur.next  # 新节点的next指向原节点的next

            cur.next = new_node       # 原节点的next指向新节点

            cur = new_node.next       # 跳到下一个原节点

        # 2. 处理新节点的random指针

        cur = head

        while cur:

            new_node = cur.next

            # 若原节点有random，则新节点的random是原random的next（对应新节点）

            if cur.random:

                new_node.random = cur.random.next

            cur = new_node.next  # 跳到下一个原节点

        # 3. 拆分链表：分离原节点和新节点

        old_cur = head

        new_head = head.next  # 新链表的头是原头的next

        new_cur = new_head

        while old_cur:

            # 恢复原链表的next

            old_cur.next = old_cur.next.next

            # 新链表的next指向新节点的下一个新节点（若存在）

            if new_cur.next:

                new_cur.next = new_cur.next.next

            # 双指针后移

            old_cur = old_cur.next

            new_cur = new_cur.next

        return new_head
```

我的思路是先把所有的值储存在列表中，然后排序，再一个个赋值给原节点，但是这么做取了巧，不能使用列表，而且空间复杂度是N，正确的应该是归并算法
```python
class Solution:

    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:

        if not head:

            return head

        cur = head

        val_list = []

        while cur:

            val_list.append(cur.val)

            cur = cur.next

        val_list.sort()

        cur = head

        for i in val_list:

            cur.val = i

            cur = cur.next

        return head
```
归并算法的关键是找到中间点，然后分成两半，一直分到不能在分的情况，然后两两比较，然后把分好的左右两半合并成一个，一直递归到最上方
```python
class Solution:

    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:

        if not head or not head.next:

            return head

        # 1. 计算链表长度

        length = 0

        cur = head

        while cur:

            length += 1

            cur = cur.next

        # 2. 虚拟头节点（方便处理头节点）

        dummy = ListNode(0, head)

        # 3. 自底向上拆分+合并：子链表长度从1开始，每次翻倍

        sub_len = 1

        while sub_len < length:

            prev = dummy  # 合并后的子链表的前驱节点

            cur = dummy.next  # 当前遍历到的节点

            while cur:

                # 拆分第一个子链表（长度sub_len）

                head1 = cur

                for _ in range(sub_len - 1):

                    if cur.next:

                        cur = cur.next

                    else:

                        break

                head2 = cur.next  # 第二个子链表的头

                cur.next = None  # 切断第一个子链表

                # 拆分第二个子链表（长度sub_len）

                cur = head2

                for _ in range(sub_len - 1):

                    if cur and cur.next:

                        cur = cur.next

                    else:

                        break

                next_node = None  # 下一轮拆分的起始节点

                if cur:

                    next_node = cur.next

                    cur.next = None  # 切断第二个子链表

                # 合并两个有序子链表，并接在prev后面

                merged = self.merge(head1, head2)

                prev.next = merged

                # 移动prev到合并后的子链表末尾

                while prev.next:

                    prev = prev.next

                cur = next_node  # 处理下一组子链表

            sub_len <<= 1  # 子链表长度翻倍（1→2→4→...）

        return dummy.next

    # 辅助函数：合并两个有序链表

    def merge(self, l1: ListNode, l2: ListNode) -> ListNode:

        dummy = ListNode(0)

        cur = dummy

        while l1 and l2:

            if l1.val <= l2.val:

                cur.next = l1

                l1 = l1.next

            else:

                cur.next = l2

                l2 = l2.next

            cur = cur.next

        cur.next = l1 if l1 else l2

        return dummy.next
```
这里每一行代码的作用如下：
首先，边界判断和计算链表的总长度，如果是空链表或者一个节点的链表，那自身就是有序的，直接返回即可。然后计算链表的长度是为了确定子链表拆分的次数
然后先出书哈虚拟头节点，令其指向head，设置子链表的初始长度为1，进入while 循环，且每次子链表的长度翻倍，直到最终与length相同，退出循环。
这个while循环的主要目的就是遍历整个链表，把它拆成多组“两个相邻的有序子链表”，然后合并每组为更长的有序子链表，最后把所有合并后的子链表衔接成一个新链表。
首先，prev用来记录已经合并部分的尾节点，cur记录当前拆分的起始节点，从cur开始，拆出两个长度尾sub_len的有序子链表（head1，head2），然后合并这两个子链表，把这两个子链表合并成更长的有序子链表，然后推进，把合并结果接在prev的后面，然后更新指针，把prev指向排序结果的最后一个节点，cur也更新到下一个需要合并的节点（注意这里要预先用next_node保存一下）。当把每一个sub_len长度的链表都成功拆分并且合并成有序链表之后，也就是sub_len长度翻倍了，就运行sub_len  <<= 1，这个的意思是把数字的二进制向左移动一位，在十进制中就等价于乘以2，然后进入下一个while循环
合并部分的函数就是比大小然后安排next节点。
