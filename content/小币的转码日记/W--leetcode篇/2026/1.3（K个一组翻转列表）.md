```python
class Solution:

    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:

        if not head or k == 1:

            return head

        def reverse(head,n):

            pre = head.next

            cur = head.next.next

            firstnode = pre

            nextnode = cur

            x = 2#记录cur的位置

            while x < n+1:

                nextnode = cur.next

                cur.next = pre

                pre = cur

                cur = nextnode

                x = x + 1

            head.next = pre

            firstnode.next = nextnode

        fakenode = ListNode(0)

        slow = fakenode

        fakenode.next = head

        fast = head

        y = 1#记录快指针的位置

        z = 0#记录翻转的次数

        while fast:

            if y < k:

                fast = fast.next

                y += 1

            else:

                last_node = slow.next

                reverse(slow,k)

                if z == 0:

                    new_head = fakenode.next

                slow = last_node

                fast = last_node.next

                y = 1

                z += 1

        return new_head
```
我的思路是用双指针找到可以进行翻转的k个节点，然后进行翻转，但是再定义reverse函数时，为了防止链表断裂，翻转的其实是slow指针后k个节点，并且同样为了不断裂，还要先保存fast的下一个节点，然后让最终的pre.next等于这个节点。然后就是，因为第一个翻转过程中，是不存在head之前的节点的，使用要定义一个fakenode，没用的节点，让它设为slow的开始，并把new_head设为翻转后fakenode的下一个节点即可
