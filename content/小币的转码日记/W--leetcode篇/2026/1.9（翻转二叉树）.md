```python
class Solution:

    def invertTree(self, root: Optional[TreeNode]) -> Optional[TreeNode]:

        if not root:

            return

        if not root.left and not root.right:

            return root

        self.invertTree(root.left)

        self.invertTree(root.right)

        root.right,root.left = root.left,root.right

        return root
```
