class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        is_vallid = True
        def isvalidbst(node):
            nonlocal is_vallid
            if is_vallid == False:
                return 
            if not node.left and not node.right:
                return
            elif not node.left and node.right:
                isvalidbst(node.right)
                min_node = self.find_max_left(node.right)
                if min_node.val <= node.val:
                    is_vallid = False
                    return
                else:
                    return
            elif node.left and not node.right:
                isvalidbst(node.left)
                max_node = self.find_max_right(node.left)
                if max_node.val >= node.val:
                    is_vallid = False
                    return
                else:
                    return
            else:
                isvalidbst(node.left)
                isvalidbst(node.right)
                min_node = self.find_max_left(node.right)
                max_node = self.find_max_right(node.left)
                if min_node.val <= node.val or max_node.val >= node.val:
                    is_vallid = False
                    return
                else:
                    return
        isvalidbst(root)
        return is_vallid
    def find_max_left(self,node):
        if not node.left:
            return node
        else:
            return self.find_max_left(node.left)
    def find_max_right(self,node):
        if not node.right:
            return node
        else:
            return self.find_max_right(node.right)
```
我的思路就是写两个找子树的最小值和最大值的函数，如果有左子树，就与左子树的最大值比较，有右子树，就与右子树的最小值比较，然后递归的结束条件就要么以及发现不是BST了，要么就找到空节点。

class Solution:
    def kthSmallest(self, root: Optional[TreeNode], k: int) -> int:
        res = []
        def inOrder(node):
            if not node:
                return
            inOrder(node.left)
            res.append(node.val)
            inOrder(node.right)
        inOrder(root)
        return res[k-1]
```

```python
class Solution:
    def kthSmallest(self, root: Optional[TreeNode], k: int) -> int:
        need_node = None
        x = 0
        def inOrder(node):
            nonlocal need_node,x,k
            if need_node or not node:
                return 
            inOrder(node.left)
            x += 1
            if x == k:
                need_node = node
                return
            inOrder(node.right)
        inOrder(root)
        return need_node.val
```
这版的空间少用了一点，但是也没少到哪里，思路就是用一个x作为标签，中序遍历，当第一次找到底部时，为x赋值为0，然后依此往上相加，中序遍历，空间没少多少是因为还是要中序遍历，栈打开的层数基本一致
