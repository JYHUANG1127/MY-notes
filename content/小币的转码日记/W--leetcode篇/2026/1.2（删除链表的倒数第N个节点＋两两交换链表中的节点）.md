```python
class Solution:

    def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:

        if not head or not head.next:

            head = None

            return head

        first_index = head

        second_index = head

        steps = 0

        while first_index.next:

            if steps != n:

                steps += 1

            else:

                second_index = second_index.next

            first_index = first_index.next

        if steps < n:

            return head.next

        elif second_index.next:

            nextnode = second_index.next.next

        else:

            nextnode = None

        second_index.next = nextnode

        return head
```
我的思路就是快慢指针，先让快指针走n+1步，然后慢指针开始一起走，直到快指针到达尾端，然后删去慢指针的下一个节点。这里要注意的就是边界情况，有删去head的，也有删去尾端的，都要注意一下

```python
class Solution:

    def swapPairs(self, head: Optional[ListNode]) -> Optional[ListNode]:

        if not head or not head.next:

            return head

        first = head

        second = head.next

        new_head = second

        while second.next and second.next.next:

            in_listnode = second.next

            next_pair = in_listnode.next

            second.next = first

            first.next = next_pair

            first = in_listnode

            second = next_pair

        remaining = second.next

        second.next = first

        first.next = remaining

        return new_head
```

```python
class Solution:

    def swapPairs(self, head: Optional[ListNode]) -> Optional[ListNode]:

        # 递归终止：无节点 / 只有一个节点，直接返回

        if not head or not head.next:

            return head

        # 取出当前要交换的两个节点

        node1 = head

        node2 = head.next

  

        # 1. 交换当前对：node2指向node1，node1指向“剩余链表交换结果”

        node1.next = self.swapPairs(node2.next)

        node2.next = node1

  

        # 2. 返回交换后的头节点（node2）

        return node2
```
