```python
# 定义双向链表节点（存储key、value，以及前后指针）

class DLinkedNode:

    def __init__(self, key=0, value=0):

        self.key = key    # 存key是为了删除尾节点时，能同步删除哈希表中的映射

        self.value = value

        self.prev = None  # 前驱指针

        self.next = None  # 后继指针
class LRUCache:

    def __init__(self, capacity: int):

        self.cache = {}          # 哈希表：key → 双向链表节点（O(1)查找）

        self.capacity = capacity # 缓存最大容量

        self.size = 0            # 当前缓存中节点的数量

        # 伪头、伪尾节点（避免处理空链表的边界情况）

        self.head = DLinkedNode()

        self.tail = DLinkedNode()

        self.head.next = self.tail

        self.tail.prev = self.head

  

    # 辅助方法：将节点添加到伪头节点的后面（标记为“最近使用”）

    def _add_to_head(self, node):

        node.prev = self.head

        node.next = self.head.next

        self.head.next.prev = node

        self.head.next = node

  

    # 辅助方法：从链表中移除指定节点

    def _remove_node(self, node):

        node.prev.next = node.next

        node.next.prev = node.prev

  

    # 辅助方法：将节点移动到伪头节点后面（更新为“最近使用”）

    def _move_to_head(self, node):

        self._remove_node(node)   # 先从原位置移除

        self._add_to_head(node)   # 再添加到头部

  

    # 辅助方法：移除伪尾节点的前驱节点（最久未使用的节点），并返回该节点

    def _remove_tail(self):

        node = self.tail.prev

        self._remove_node(node)

        return node

  

    def get(self, key: int) -> int:

        if key not in self.cache:

            return -1  # key不存在，返回-1

        # key存在：获取节点，移到头部（标记为最近使用），返回value

        node = self.cache[key]

        self._move_to_head(node)

        return node.value

  

    def put(self, key: int, value: int) -> None:

        if key not in self.cache:

            # 新增节点

            new_node = DLinkedNode(key, value)

            self.cache[key] = new_node  # 哈希表记录映射

            self._add_to_head(new_node) # 加到头部（最近使用）

            self.size += 1

            # 若超出容量，移除最久未使用的节点（伪尾的前驱）

            if self.size > self.capacity:

                removed_node = self._remove_tail()

                del self.cache[removed_node.key]  # 同步删除哈希表中的映射

                self.size -= 1

        else:

            # key已存在：更新value，并移到头部（标记为最近使用）

            node = self.cache[key]

            node.value = value

            self._move_to_head(node)
```

豆包的思路是利用哈希表＋==双向链表==（md还有这种东西，我怎么从来不知道），在这个双向链表中一共包含4个内容，首先key节点的标志，value节点代表的值，还有prev上一个节点，next下一个节点
然后在定义这个LRU数据结构时，它一共包含5个内容，首先是初始的哈希表为空（用来以复杂度O(1)来找到节点，储存的key就是节点的key和整个节点（包括它的key，value，prev，next）），然后是capacity最大容量，初始的size是0，且定义了伪头和伪尾，都是双向链表，且self.head的next是self.tail,self.tail的prev是self.head
然后是辅助函数add，把节点添加到伪节点的后面，辅助函数remove，把节点从这个链表中删除。这两个辅助函数的目的都是为了把刚刚操作的节点移到首段，因为我们把首节点（也就是self.head.next）标记为最近使用的节点，方便后续内存满了之后删除最尾端的节点，也就是self.tail.prev
然后实现get函数作用，先判断key是否在哈希表中，如果没有，就返回-1，如果有，就先把这个节点移到首端，然后返回节点的值
然后实现put函数作用，同样先判断这个key是否已经储存在了哈希表中，如果没有，就新生成一个双链表的节点，然后插入首端，加入哈希表中，并且size＋1，如果size大于最大容量，还要把最后一个节点删去，注意在这里哈希表中的节点也要同步删去，如果key已经存在与哈希表中，就直接更新节点的值就行，并把它移到首端
这里可以以双端节点的key作为哈希表的key 而不用担心两个key相同但是value不同的原因是要储存节点进入哈希表就必须通过put函数，但是put函数如果遇到key相同的就只会更新这个key相同的双端节点的value，所以不会发生两个key相同但是value不同的节点同时储存进入哈希表中
