```python
class Solution:

    def mergeKLists(self, lists: List[Optional[ListNode]]) -> Optional[ListNode]:

        import heapq

        from typing import List, Optional

        # 虚拟头节点：简化结果链表的构建

        dummy = ListNode(0)

        cur = dummy

        # 初始化小顶堆（Python的heapq是小顶堆）

        heap = []

        # 遍历每个链表，将非空链表的头节点加入堆

        for idx, head in enumerate(lists):

            if head:

                # 存 (节点val, 索引, 节点)：索引用于避免val相同时节点对象无法比较的问题

                heapq.heappush(heap, (head.val, idx, head))

        # 循环取堆顶（最小节点），构建结果链表

        while heap:

            # 弹出堆中val最小的节点

            val, idx, node = heapq.heappop(heap)

            # 将该节点接到结果链表上

            cur.next = node

            cur = cur.next

            # 如果该节点有下一个节点，将下一个节点加入堆

            if node.next:

                heapq.heappush(heap, (node.next.val, idx, node.next))

        # 返回合并后的链表头（虚拟头节点的下一个）

        return dummy.next
```

```python
class Solution:

    def mergeKLists(self, lists: List[Optional[ListNode]]) -> Optional[ListNode]:

        # 边界条件：空列表直接返回None

        if not lists:

            return None

        # 分治终止条件：只剩1个链表，直接返回

        if len(lists) == 1:

            return lists[0]

        # 1. 拆分：找中间点，拆成左右两部分

        mid = len(lists) // 2

        left = self.mergeKLists(lists[:mid])   # 递归处理左半部分

        right = self.mergeKLists(lists[mid:]) # 递归处理右半部分

        # 2. 归并：合并左右两个有序链表

        return self.mergeTwoLists(left, right)

    # 辅助函数：合并两个有序链表（基础子问题）

    def mergeTwoLists(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:

        # 虚拟头节点，简化合并逻辑

        dummy = ListNode(0)

        cur = dummy

        # 双指针遍历两个链表，按序拼接

        while l1 and l2:

            if l1.val < l2.val:

                cur.next = l1

                l1 = l1.next

            else:

                cur.next = l2

                l2 = l2.next

            cur = cur.next

        # 拼接剩余的节点（l1或l2可能还有未遍历的节点）

        cur.next = l1 if l1 else l2

        return dummy.next
```
