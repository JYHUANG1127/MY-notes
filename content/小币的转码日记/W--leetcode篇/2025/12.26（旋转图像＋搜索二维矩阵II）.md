class Solution:

    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:

        if not matrix or not matrix[0]:

            return False

        m = len(matrix)    # 行数

        n = len(matrix[0]) # 列数

        x, y = 0, n - 1    # 从右上角开始（行0，列n-1）

        while x < m and y >= 0:

            current = matrix[x][y]

            if current == target:

                return True

            elif current > target:

                y -= 1  # 目标在左侧，列左移

            else:

                x += 1  # 目标在下方，行下移

        return False
