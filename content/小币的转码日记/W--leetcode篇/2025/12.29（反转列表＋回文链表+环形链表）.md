class Solution:

    def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:

        prev = None  # 前驱节点初始化为空

        curr = head  # 当前节点从原链表头开始

        while curr:

            next_node = curr.next  # 先保存当前节点的下一个节点

            curr.next = prev       # 反转当前节点的指针（指向前驱）

            prev = curr            # 前驱节点后移

            curr = next_node       # 当前节点后移

        return prev  # 遍历结束后，prev就是新的链表头

下面的是豆包的最优解思路，它的思路是使用快慢指针先找中点，如果是奇数长度就跳过中间节点。并且在fast移动时顺便反转前半部分的链表。然后进行比较即可。它利用双指针判断中点然后翻转前半部分的链表的思路是，slow和fast同时为head，然后fast走两步，slow走一步，然后利用slow翻转，用fast判断是否到终点。
```python
class Solution:
    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        # 边界情况：空/单节点直接返回True
        if not head or not head.next:
            return True
        
        # 步骤1：快慢指针找中点（合并反转后半部分的初始化）
        slow = fast = head
        prev = None  # 用于反转后半部分的前驱指针
        # 快慢指针移动时，直接反转前半部分（仅适用于“比较时不需要保留原链表”的场景）
        # 注：此写法会修改原链表，但能减少一次遍历+节省空间
        while fast and fast.next:
            fast = fast.next.next
            # 同时反转前半部分（替代“后续反转后半部分”的步骤）
            next_slow = slow.next
            slow.next = prev
            prev = slow
            slow = next_slow
        
        # 步骤2：处理奇数长度（跳过中间节点）
        if fast:
            slow = slow.next
        
        # 步骤3：比较（prev是反转后的前半部分，slow是后半部分）
        while slow:
            if prev.val != slow.val:
                return False
            prev = prev.next
            slow = slow.next
        
        return True
```

        has_cycle = False

        x = head

        if x == None:

            return has_cycle

        if x.next == None:

            return has_cycle

        L = []

        while x:

            if x not in L:

                L.append(x)

                x = x.next

            else:

                has_cycle = True

                break

        return has_cycle

豆包的思路还是快慢指针，快指针走的速度是慢指针的两倍，让快指针先走，如果有环，快指针就会在环内追上慢指针，如果无环，快指针就会遇到none，然后结束
```python
class Solution:

    def hasCycle(self, head: Optional[ListNode]) -> bool:

        # 边界情况：空链表或单节点链表（无环）

        if not head or not head.next:

            return False

        # 初始化快慢指针（慢指针从head出发，快指针从head.next出发，避免初始相等）

        slow = head

        fast = head.next

        # 循环条件：快慢指针不相等

        while slow != fast:

            # 快指针走到末尾（无环）

            if not fast or not fast.next:

                return False

            # 慢指针走1步，快指针走2步

            slow = slow.next

            fast = fast.next.next

        # 退出循环说明快慢指针相遇（有环）

        return True
```
