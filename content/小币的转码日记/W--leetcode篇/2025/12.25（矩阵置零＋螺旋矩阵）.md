class Solution:

    def setZeroes(self, matrix: List[List[int]]) -> None:

        """

        Do not return anything, modify matrix in-place instead.

        """

        def has_zero(L:list):

            index_zero = []

            for i in range(len(L)):

                if L[i] == 0:

                    index_zero.append(i)

            return index_zero

        all_index = []

        for i in range(len(matrix)):

            if len(has_zero(matrix[i])) != 0:

                all_index.extend(has_zero(matrix[i]))

                matrix[i] = [0]*len(matrix[i])

        if len(all_index) != 0:

            for i in all_index:

                for j in range(len(matrix)):

                    matrix[j][i] = 0

豆包给的算法是边界收缩法
```python
class Solution:

    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:

        if not matrix or not matrix[0]:

            return []

        m, n = len(matrix), len(matrix[0])

        top, bottom = 0, m - 1

        left, right = 0, n - 1

        res = []

        while True:

            # 1. 遍历上边界：左→右

            for j in range(left, right + 1):

                res.append(matrix[top][j])

            top += 1

            if top > bottom:

                break

            # 2. 遍历右边界：上→下

            for i in range(top, bottom + 1):

                res.append(matrix[i][right])

            right -= 1

            if left > right:

                break

            # 3. 遍历下边界：右→左

            for j in range(right, left - 1, -1):

                res.append(matrix[bottom][j])

            bottom -= 1

            if top > bottom:

                break

            # 4. 遍历左边界：下→上

            for i in range(bottom, top - 1, -1):

                res.append(matrix[i][left])

            left += 1

            if left > right:

                break

        return res
```
这个算法里是遍历边界，因为是顺时针，所以按照右-下-左-上的顺序遍历边界，最关键的就是while语句中的介绍条件，就是当满足top>bottom或者left>right时都可以跳出while语句
